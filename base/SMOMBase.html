# 单表

  • 在对应项目下新建文件夹加上s结尾。
  • 创建实体类
  • 实体必须标记为根实体（RootEntity）/子实体（ChildEntity），序列化（Serializable）特性[RootEntity,Serializable]。
  • 配置查询面板，框架自带的查询标记特性：[CriteriaQuery]；自定义的查询标记特性：[ConditionQueryType(typeof(XXXXXCriteria))]
  • 通过代码段创建属性，pfp 创建属性面板
    • PFProperty：一般属性（常用）
    • PFPropertyReference：非空引用属性（常用）
    • PFPropertyReferenceNullable：可空引用属性（常用）
    • PFViewProperty：视图属性，可以有应用属性带出属性值（常用）
    • PFPropertyList：列表属性（常用）
  • 实体配置，继承自EntityConfig<当前实体类>
  • 配置元数据映射实体对应的数据库表
using System;

# 验证规则

  • 有三种验证规则：标准规则、运行时规则、预编译规则
  • 标准规则：标准规则是在实体配置(EntityConfig)中重写AddValidations()方法实现规则验证;标准规则不可在实体元数据功能中配置，而是代码编写完成运行程序立马生效,不需要更新实体元数据。
  • 运行时规则：通过在实体元数据模块进行运行时配置的规则(给实体添加特性,如：[Required])，如字段长度、非空等验证。运行时规则完全通过配置实现。需要在实体元数据中进行更新才会生效。
  • 预编译规则:通过C#代码的方式实现验证规则，然后在实体元数据功能进行更新并配置启用;包含的规则子类：实体规则(继承EntityRule)、非重复规则(继承NotDuplicateRule)、删除被引用规则(继承NoReferencedRule)。该规则也需要在实体元数据中进行更新才会生效。
    • 使用场景：常用来实现特定的验证提示与稍复杂的业务验证规则。

# 标准规则

  • 标准规则可以实现的验证：
    • 非空验证，使用：rules.AddRule(属性名, new RequiredRule())
    • 非重复验证，单个和多个属性的非重复验证；
    internal class SupplierSmallConfig : EntityConfig<SupplierSmall>
{
    protected override void ConfigMeta()
    {
        Meta.MapTable("LIB_SUPPLIER_SMALL").MapAllProperties();
        Meta.EnablePhantoms();
    }
    protected override void AddValidations(ValidationRules rules)
    {
      base.AddValidations(rules);
      //单个属性非重复验证
      rules.AddRule(属性名, new NotDuplicateRule());
    }
}

    protected override void AddValidations(ValidationRules rules)
{
  base.AddValidations(rules);
  //多个属性非重复验证
  rules.AddRule(new NotDuplicateRule()
  {
      Properties =
      {
          属性名1,
          属性名2
      },
      MessageBuilder = (e) =>
      {
          return "验证不通过的提示信息".L10N();
      }
  });
}
    • 长度验证，包括最小和最大长度的设置：
    protected override void AddValidations(ValidationRules rules)
{
  base.AddValidations(rules);
  rules.AddRule(属性名, new StringLengthRangeRule() { Min = 2, Max = 40 });
}
    • 最大最小值验证：
    protected override void AddValidations(ValidationRules rules)
{
  base.AddValidations(rules);

    • 正则表达式验证:
    protected override void AddValidations(ValidationRules rules)
{
  base.AddValidations(rules);
  rules.AddRule(属性名1, new RegexMatchRule()
  {
    Regex = new Regex(@"^((\d{3}-\d{8}|\d{4}-\d{7,8})|(0?(13|14|15|17|18|19)[0-9]{9}))$"),

    • 实体验证：
    protected override void AddValidations(ValidationRules rules)
{
  base.AddValidations(rules);
  rules.AddRule(new HandlerRule()//等同于预编译规则的 new EntityRule<实体类>()
  {
    Handler = (o, e) =>
    {    //这里可以进行查数据库进行操作，当前实体的数据不能满足验证判断要求时可查数据库判断
    var reader = o.CastTo<实体类>();
    if (判断的条件)
      e.BrokenDescription = “具体的验证提示信息".L10nFormat(reader.Name);
    }
  });
}

# 运行时规则

  • 通过在实体元数据模块进行运行时配置的规则，如字段长度、非空等验证。运行时规则完全通过配置实现。
  • 运行时规则还可以通过在实体属性中标记特性来实现的，完成后要进行实体元数据的初始化，在实体元数据菜单的实体规则可以配置启用和禁用，默认初始化完成后都是启用状态。
  • 常用的规则特性有：
    • [Required]：非空验证
    • [NotDuplicate]：非重复验证
    • [MaxLength(10)]：最大长度验证
    • [MinLength(3)]：最小长度验证
    • [MaxValue(100)]：最大值验证
    • [MinValue(1)]：最小值验证

# 预编译规则

  • 通过C#代码的方式实现验证规则，然后在实体元数据功能进行更新并配置启用。 包含的规则子类：实体规则(继承EntityRule)、非重复规则(继承NotDuplicateRule)、删除被引用规则(继承NoReferencedRule)。

  • 在服务端新建一个规则类，命名以Rule结尾。

  • 实体规则继承EntityRule<对应操作的实体>，完成后对该规则进行特性的标记

  • 重写验证规则的方法，在验证规则方法中实现具体的业务逻辑

  • 初始化元数据，完成后在界面中进行验证

    • 使用场景：常用来实现特定的验证提示与稍复杂的业务验证规则。
    • 继承EntityRule,在这个规则中可以实现所有规则。
    namespace SIE.Lib.Suppliers
{
  /// <summary>
  /// 同一供应商下，不可存在重复的物料
  /// </summary>
  /// 因为不在一个实体里面，不知道能否校验成功
  [DisplayName("同一供应商下，不可存在重复的物料")]
  [Description("同一供应商下，不可存在重复的物料")]
  public class SupplierSmallRuler : NotDuplicateRule<SupplierSmall>
  {
      public SupplierSmallRuler()
      {
          Properties.Add(SupplierSmall.SupplierIdIdProperty);
          Properties.Add(SupplierSmall.MaterialIdProperty);
      }
  }

  /// <summary>
  /// 自定义规则校验实体EntityRule
  /// </summary>
  [DisplayName("手机号规则校验")]
  [Description("手机号正则表达式规则校验")]
  public class SupplierRefPhoneRuler : EntityRule<Supplier>
  {
      public SupplierRefPhoneRuler() { 
          Property = Supplier.ContactPhoneProperty;//校验触发条件，PhoneProperty属性
          ConnectToDataSource = false;//ConnectToDataSource默认值为false，如果是在Validate方法中有差数据库的操作，则需要把该属性设置为true
      }
      protected override void Validate(IEntity entity, RuleArgs e)
      { 
          var reader = entity as Supplier;
          //手机正则表达式校验逻辑实现
          if (!reader.ContactPhone.IsNullOrWhiteSpace())
          {
              string phoneRule = @"^((\d{3}-\d{8}|\d{4}-\d{7,8})|(0?(13|14|15|17|18|19)[0-9]{9}))$";
              Regex phoneRegx = new Regex(phoneRule);
              var matches = phoneRegx.IsMatch(reader.ContactPhone);
              if (!matches)
              {
                  //校验失败需要赋值给BrokenDescription
                  e.BrokenDescription = "手机号[{0}]格式不正确".L10nFormat(reader.ContactPhone);
              }
          }
      }
  }
}
    • 校验不允许删除示例：
    • 调用控制器的方法实现：RT.Service.Resolve<具体的控制器>().方法名(参数)
    /// <summary>
/// 单位被物料引用不允许删除
/// </summary>
[DisplayName("单位被物料引用不允许删除")]
[Description("单位被物料引用不允许删除校验逻辑")]
public class MaterialRefUnitRuler : NoReferencedRule<Employer>
{
    //构造函数
    public MaterialRefUnitRuler() { 
        Scope = EntityStatusScopes.Add | EntityStatusScopes.Update;//校验触发条件，新增或者修改时校验，Scope如果不指定，默认值是AddOrUpdate
        ConnectToDataSource = true; //是否需要链接数据库
    }
    //重写验证规则的方法,存在物料引用则提示不能删除
    protected override void Validate(IEntity entity, RuleArgs e)
    { 
      var reader = entity as Employer;
      if(RT.Service.Resolve<MaterialController>().IsExistMaterial(reader.Id))//获取物料里是否存在当前单位
      {
        e.BrokenDescription = "单位[{0}]被物料引用不允许删除".L10nFormat(reader.Name);
      }
    }
}
    • 说明：控制必须建在服务端，且是部分类，继承自DomainController，控制器的方法必须是虚方法(virtual)
    public partial class MaterialController : DomainController
{
  public virtual bool IsExistMaterial(Guid employerId)
  {
    if(employerId <= 0 ){
      throw new ArgumentNullException(nameof(employerId));
    }
    var query = new Query<Material>();
    query.Where(x => x.EmployerId == employerId);
    return query.Count() > 0;
  }
}
    • 非重复验证使用示例
    • 当设置一个字段时，则只对实体的一个字段值在数据库中进行验证，如果设置多个字段值时，则将实体多个字段值在数据库中验证。
    • 在构造函数中，通过Properties.Add方法同，添加判断重复的字段，可以添加一个或者多个字段，例如编码字段Code不能重复，则只需要添加Code字段。如多个字段时，则是将多个字段的值合起来一起判断。

}

# 后端验证通用说明

  • 标准验证AddValidations和预编译验证(在服务端编写验证规则类)是两种不一样的实现方式，同一逻辑用两种方式验证的效果是一样的，只是其中一种是代码所见即所得，一种是可配置；
  • 在项目上对应的验证逻辑，能用标准规则验证就不要使用其他的验证，主要原因是实体元数据更新执行效率比较低；
  • 非重复验证尽量使用框架非重复的子类进行验证，不要在实体规则中进行验证，原因是写的逻辑比较多，还有可能写漏，用框架的非重复验证会简单很多，也避免一些不必要的bug出现；
  • 被引用不允许删除尽量使用框架的，原因与非重复验证一致；
  • String类型的长度验证，框架默认配置的长度是20，如果有些字段需要设置为其他的长度限制，需要自己手动设置，如果框架string类型的长度限制都需要调整，可以在配置文件中配置DefaultFieldMaximumLength。

# 提交事件

  • 提交事件建在服务端，而且在第一次使用或者修改了对应的事件，都必须初始化元数据才会生效。
  • 提交事件使用场景：针对某个菜单功能的通用处理逻辑，可用在提交事件中进行处理。
  • 提交事件分为提交前和提交后事件：
    • 提交前事件，是在保存这个实体对象之前调用的；
    • 提交后事件，是在保存这个实体对象之后调用的；

# 提交前事件

  • 在服务端新建一个类，命名以OnSubmitting结尾，继承OnSubmitting，OnSubmitting后面跟对应的操作实体,
  • SuppliersOnSubmitting报错需要根据提示选择实现抽象类

# 提交后事件

  • 在服务端新建一个类，命名以OnSubmitted结尾，继承OnSubmitted，OnSubmitted后面跟对应的操作实体
using SIE.Domain;
using System.Linq;
using System;
using System.Collections.Generic;
using System.Text;

namespace SIE.Lib.Suppliers
{
    public class SuppliersOnSubmitted : OnSubmitted<Supplier>
    {
        /// <summary>
        /// 供应商提交后事件处理
        /// </summary>
        /// <param name="entity"></param>
        /// <param name="e"></param>
        protected override void Invoke(Supplier entity, EntitySubmittedEventArgs e)
        {
            //判断类型是新增或者更新
            if (e.Action == SubmitAction.Insert || e.Action == SubmitAction.Update) {
                SupplierAddress supAddress = entity.GetProperty(SupplierAddressExtention.SupAddressProperty);
                if (supAddress != null && supAddress.SupplierId == 0) {
                    supAddress.SupplierId = entity.Id;
                    supAddress.Supplier = entity;
                    var existingAddress = RT.Service.Resolve<SuppliersController>().GetSupAddress(entity.Id);
                    if (existingAddress == null)
                    {
                        // 不存在，设为新建状态
                        supAddress.PersistenceStatus = PersistenceStatus.New;
                    }
                    RF.Save(supAddress);
                }
                else if (supAddress !=null && supAddress.SupplierId > 0)
                {
                    RF.Save(supAddress);
                }
            }
            if (e.Action == SubmitAction.Delete) { 
                var address = RT.Service.Resolve<SuppliersController>().GetSupAddress(entity.Id);
                if (address != null) {
                    address.PersistenceStatus = PersistenceStatus.Deleted;
                    RF.Save(address);
                }
            }

        }
    }

}

# 提交事件注意事项

  • 提交前、提交后事件使用要生效，必须进行实体元数据更新，状态为可用；
  • 每次修改了提交前或者提交后事件，也必须进行实体元数据的更新；
  • 提交事件后台代码删除，在实体元数据对应提交事件子列表的数据不会删除，需要手动去提交事件子页签将对应数据禁用，或者是在后台数据库将数据删除。
    • 提交事件的表为：MDA_ENTITY_SUBMIT_EVENT，通过DISCRIMINATOR区分为提交前后事件；
    • DISCRIMINATOR=‘A’，为提交前事件；
    • DISCRIMINATOR=‘B’，为提交后事件；
  • 提交事件后台的类名修改了，修改之前通过“更新”生成的提交事件对应的数据程序不会删除，会重新再生成一笔数据；也就是提交事件在“更新”之后修改了类名，需要把修改类名之前的数据禁用或者从数据库中删除。
  • 如果是通过DB操作的数据，或者是批量保存，不会执行对应的验证规则。

# API接口

  • 数据推送接口的定义：
  • 接口协议：RestFul API
  • 方法名：Login
  • 请求方式（Method）：POST
  • 参数形式（Content Type）：application/json
  • 描述：第三方用户调用此接口获取数据
  • 参数：
{
"ApiType": "AuthenticationController",
  "Parameters": [
    {
      "Value": "str"
    },
    {
      "Value": "str"
    }
  ],
  "Method": "Login",
  "Context": {}
}
  • 参数描述:参数内容需符合JSON格式，JSON串中的键仅允许26个小写英文字母、阿拉伯数字和下划线，且必须以小写字母开头，当键由多个单词组成时，单词与单词之间通过下划线进行连接，如未特别说明，本规范中定义的各字段的命名不能修改，最大封装Json包100KB。
    • ApiType：控制器名称
    • Parameters：参数
    • Method：方法名称
    • Context：上下文内容
  • 返回值：
//成功返回值
{
  "Success": true,
  "Message": null,
  "Result": 0,
  "Context": {
    "Ticket": "XXX"
  }
}
//错误返回值
{
  "Success": false,
  "Message": “错误信息”,
  "Result": 1,
  "Context": {
    "Ticket": ""
  }
}
  • 返回值描述:
    • Json格式，
    • Success是否成功：true成功，false失败；
    • Message错误信息；
    • Result返回状态，0成功，1失败；
    • Context内容：Ticket是凭证，在登录成功的时候MES会生成一个Ticket，此Ticket在非匿名方法的时候必须作为参数传入MES。
  • 备注:每个Ticket有效期默认为30分钟。

# 框架Api的使用

  • 说明：
    • api方法与普通方法的区别，api方法多了api标记的特性
    • ApiService：api开放接口标记
    • ApiReturn：api开放接口返回值
    • ApiParameter：api开放接口参数说明
  • 方法标记了ApiService，在运行host或者是部署成服务后，可以在api中查到对应的方法、请求格式和返回值，其他系统也可访问调用对应服务的api方法。
  • 查看写好的api接口：查看api，以本地调试为例，在SIE.WebApiHost中将写api方法的工程引用进来。
// <summary>
/// 查询修改或者新增单位
/// </summary>
/// <param name="request"></param>
[ApiService("修改或者新增单位")]//api标记
[return: ApiReturn("修改或者新增单位")]//api返回值
public virtual void QueryOrAddUnit([ApiParameter("查询对象")] QueryOrAddUnitRequest request)//ApiParameter标记api参数
{
    var queryRequest = new QueryOrAddUnitRequest()
    {
        Code = request.Code,
        Name = request.Name,
        Type = request.Type,
        UnitPrecision = request.UnitPrecision
    };
    //var Unit = GetById<Employer>(queryRequest.Code);
    var query = Query<Employer>();
    var employers = query.Where(p => p.Code.Contains(queryRequest.Code)).ToList();
    if (employers != null)
    {
        foreach (var employer in employers)
        {
            employer.Name = queryRequest.Name;
            employer.Type = queryRequest.Type;
            employer.UnitPrecision = queryRequest.UnitPrecision;
        }
        RF.Save(employers);//保存
    }
    else {
        var NewEmployer = new Employer()
        {
            Code = queryRequest.Code,
            Name = queryRequest.Name,
            Type = queryRequest.Type,
            UnitPrecision = queryRequest.UnitPrecision
        };
        RF.Save(NewEmployer);//保存
    }
}
